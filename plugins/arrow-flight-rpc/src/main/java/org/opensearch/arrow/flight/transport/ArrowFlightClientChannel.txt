package org.opensearch.arrow.flight.transport;

import org.apache.arrow.flight.FlightClient;
import org.apache.arrow.flight.FlightStream;
import org.apache.arrow.flight.Ticket;
import org.apache.arrow.vector.VectorSchemaRoot;
import org.opensearch.Version;
import org.opensearch.cluster.node.DiscoveryNode;
import org.opensearch.common.annotation.PublicApi;
import org.opensearch.core.action.ActionListener;
import org.opensearch.core.common.bytes.BytesReference;
import org.opensearch.transport.NodeNotConnectedException;
import org.opensearch.transport.TcpChannel;
import org.opensearch.transport.Transport;
import org.opensearch.transport.TransportException;
import org.opensearch.transport.TransportRequest;
import org.opensearch.transport.TransportRequestOptions;
import org.opensearch.transport.TransportResponseHandler;

import java.io.IOException;
import java.net.InetSocketAddress;
import java.util.Optional;
import java.util.concurrent.atomic.AtomicBoolean;

/**
 * TcpChannel implementation for sending Arrow Flight requests.
 *
 * @opensearch.api
 */
@PublicApi(since = "1.0.0")
public class ArrowFlightClientChannel implements TcpChannel, Transport.Connection {
    private final DiscoveryNode node;
    private final FlightClient flightClient;
    private final ChannelStats stats = new ChannelStats();
    private volatile boolean isOpen = true;
    private final AtomicBoolean sendMessageCalled = new AtomicBoolean(false); // Test hook

    public ArrowFlightClientChannel(DiscoveryNode node, FlightClient flightClient) {
        this.node = node;
        this.flightClient = flightClient;
    }

    @Override
    public boolean isServerChannel() {
        return false;
    }

    @Override
    public String getProfile() {
        return "arrow_flight";
    }

    @Override
    public InetSocketAddress getLocalAddress() {
        return new InetSocketAddress("localhost", 0);
    }

    @Override
    public InetSocketAddress getRemoteAddress() {
        return node.getAddress().address();
    }

    @Override
    public void sendMessage(BytesReference reference, ActionListener<Void> listener) {
        sendMessageCalled.set(true);
        try {
            Ticket ticket = new Ticket(reference.toBytesRef().bytes);
            try (FlightStream stream = flightClient.getStream(ticket)) {
                VectorSchemaRoot root = stream.getRoot();
                // Process stream and pass to response handler
                while (stream.next()) {
                    // Process each batch if needed
                    System.out.println("Received batch with " + root.getRowCount() + " rows");
                }
                // Find the response handler
                long requestId = reference.getLong(0); // Assume requestId is encoded in the bytes
                TransportResponseHandler<VectorSchemaRoot> handler = (TransportResponseHandler<VectorSchemaRoot>)
                    flightClient.getResponseHandlers().remove(requestId);
                if (handler != null) {
                    handler.handleResponse(root);
                }
                listener.onResponse(null);
            }
        } catch (Exception e) {
            listener.onFailure(new RuntimeException("Failed to send request", e));
        }
    }

    @Override
    public void addConnectListener(ActionListener<Void> listener) {
        listener.onResponse(null);
    }

    @Override
    public void addCloseListener(ActionListener<Void> listener) {
        listener.onResponse(null);
    }

    @Override
    public boolean isClosed() {
        return !isOpen;
    }

    @Override
    public ChannelStats getChannelStats() {
        return stats;
    }

    @Override
    public boolean isOpen() {
        return isOpen;
    }

    @Override
    public void close() {
        isOpen = false;
        flightClient.close();
    }

    @Override
    public <T> Optional<T> get(String name, Class<T> clazz) {
        return Optional.empty();
    }

    // Transport.Connection methods
    @Override
    public DiscoveryNode getNode() {
        return node;
    }

    @Override
    public void sendRequest(long requestId, String action, TransportRequest request, TransportRequestOptions options)
        throws IOException, TransportException {
        if (!isOpen) {
            throw new NodeNotConnectedException(node, "Connection is closed");
        }
        // Serialize request using NativeOutboundHandler (handled by TransportService)
        // The actual serialization happens in NativeOutboundHandler
        BytesReference serialized = serializeRequest(requestId, action, request, options);
        sendMessage(serialized, ActionListener.wrap(
            v -> {},
            e -> {
                TransportResponseHandler<VectorSchemaRoot> handler =
                    (TransportResponseHandler<VectorSchemaRoot>) flightClient.getResponseHandlers().remove(requestId);
                if (handler != null) {
                    handler.handleException(new TransportException("Failed to send request", e));
                }
            }
        ));
    }

    @Override
    public Version getVersion() {
        return node.getVersion();
    }

    public boolean wasSendMessageCalled() {
        return sendMessageCalled.get();
    }

    // Placeholder for serialization (handled by NativeOutboundHandler in practice)
    private BytesReference serializeRequest(long requestId, String action, TransportRequest request, TransportRequestOptions options) {
        // In practice, this is done by NativeOutboundHandler
        // For simplicity, return a dummy BytesReference with requestId
        byte[] bytes = new byte[8];
        for (int i = 0; i < 8; i++) {
            bytes[i] = (byte) (requestId >>> (i * 8));
        }
        return null;// BytesReference.bytes(bytes).;
    }
}
